Dán URL Web App triển khai từ Google Apps Script của trang tính vào đây.

📋 Hướng dẫn tạo kết nối 1 lần (Mất 30 giây):
  1. Mở file Google Sheet: Nhấn vào đây để mở Sheet
  2. Trên menu trang tính, chọn Tiện ích mở rộng (Extensions) > Apps Script.
  3. Xóa hết mã có sẵn và dán đoạn mã bên dưới vào.
  4. Nhấn Triển khai (Deploy) > Tùy chọn triển khai mới (New deployment) > Chọn loại là Ứng dụng web (Web app).
  5. Tại mục Ai có quyền truy cập (Who has access), chọn Bất kỳ ai (Anyone) > Nhấn Triển khai (Deploy).
  6. Sao chép URL ứng dụng web nhận được và dán vào ô bên trên!
Mã Google Apps Script (AppsScript.gs): 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Nếu trang tính chưa có tiêu đề cột thì tạo
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Thời gian", "Mã Barcode / QR", "Tên Model / Ghi chú", "Số lượng", "Thiết bị quét"]);
      sheet.getRange(1, 1, 1, 5).setFontWeight("bold").setBackground("#f3f4f6");
    }
    
    // Nhận danh sách mảng các mã hoặc 1 mã đơn lẻ
    var rows = Array.isArray(data) ? data : [data];
    rows.forEach(function(item) {
      sheet.appendRow([
        item.timestamp || new Date().toLocaleString("vi-VN"),
        item.code || "",
        item.note || "",
        item.quantity || 1,
        item.device || "Smart Scanner Web"
      ]);
    });
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
